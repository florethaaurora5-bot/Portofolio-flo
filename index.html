Smart Climate & Automated Cooling Servo System

Reads temperature from a DHT11 sensor and displays readings on an I2C 16x2 LCD. Automatically rotates a 90° servo flap when temperature exceeds 30°C. 

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <Servo.h>

#define DHTPIN 2
#define DHTTYPE DHT11
#define SERVO_PIN 9

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo ventServo;

void setup() {
  dht.begin();
  lcd.init();
  lcd.backlight();
  ventServo.attach(SERVO_PIN);
  ventServo.write(0); // Initial Position
}

void loop() {
  float temp = dht.readTemperature();
  lcd.setCursor(0, 0);
  lcd.print("Temp: "); lcd.print(temp); lcd.print("C");
  
  if (temp > 30.0) {
    ventServo.write(90); // Open cooling flap
    lcd.setCursor(0, 1); lcd.print("Cooling: ACTIVE ");
  } else {
    ventServo.write(0);
    lcd.setCursor(0, 1); lcd.print("Cooling: NORMAL ");
  }
  delay(2000);
}
